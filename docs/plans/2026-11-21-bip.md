```python
"""Estruturas de dados do Bip e o status de cada PR."""

from dataclasses import dataclass, field

FAILING = {"FAILURE", "ERROR"}
RUNNING = {"PENDING", "EXPECTED"}

# Ordem de exibição: primeiro o que exige ação do usuário.
ORDER = ["rebase", "failing", "changes", "running", "draft", "approved", "waiting"]
ATTENTION = {"rebase", "failing", "changes"}


@dataclass(frozen=True)
class PR:
    id: str
    repo: str  # "dono/nome"
    number: int
    title: str
    url: str
    author: str
    draft: bool
    created_at: str
    merged_at: str | None
    head: str
    base: str
    head_oid: str
    ci: str | None  # SUCCESS, FAILURE, ERROR, PENDING, EXPECTED ou None (sem checks)
    review_decision: str | None  # APPROVED, CHANGES_REQUESTED, REVIEW_REQUIRED ou None
    conflicting: bool
    last_feedback_at: str | None  # review ou comentário mais recente de outra pessoa (ISO 8601)


@dataclass(frozen=True)
class Snapshot:
    viewer: str
    review: tuple  # PRs que pedem a revisão do usuário
    mine: tuple  # PRs abertos do usuário
    merged: tuple  # PRs do usuário mergeados recentemente


@dataclass
class State:
    snapshot: Snapshot
    parent: dict = field(default_factory=dict)  # id do PR -> id do PR pai (aberto)
    rebase: dict = field(default_factory=dict)  # id do PR -> número do pai mergeado, ou None (conflito)


def status(pr, state):
    if pr.id in state.rebase:
        return "rebase"
    if pr.ci in FAILING:
        return "failing"
    if pr.review_decision == "CHANGES_REQUESTED":
        return "changes"
    if pr.ci in RUNNING:
        return "running"
    if pr.draft:
        return "draft"
    if pr.review_decision == "APPROVED":
        return "approved"
    return "waiting"


def attention_count(state):
    mine = sum(1 for pr in state.snapshot.mine if status(pr, state) in ATTENTION)
    return len(state.snapshot.review) + mine
```

`github_api.py`:
```python
"""Busca os PRs no GitHub (uma consulta GraphQL) e converte a resposta em objetos PR."""

import json
import shutil
import ssl
import subprocess
import sys
import urllib.error
import urllib.request
from datetime import timedelta
from pathlib import Path

import certifi

from model import PR, Snapshot

API_URL = "https://api.github.com/graphql"
# Onde o GitHub CLI costuma estar quando o app é aberto fora do terminal (sem o PATH do shell).
GH_PATHS = ["/opt/homebrew/bin/gh", "/usr/local/bin/gh", r"C:\Program Files\GitHub CLI\gh.exe"]

QUERY = """
query($review: String!, $mine: String!, $merged: String!) {
  viewer { login }
  review: search(query: $review, type: ISSUE, first: 50) { nodes { ...pr } }
  mine: search(query: $mine, type: ISSUE, first: 50) { nodes { ...pr } }
  merged: search(query: $merged, type: ISSUE, first: 50) { nodes { ...pr } }
}
fragment pr on PullRequest {
  id number title url isDraft createdAt mergedAt state
  headRefName baseRefName headRefOid mergeable reviewDecision
  author { login }
  repository { nameWithOwner }
  commits(last: 1) { nodes { commit { statusCheckRollup { state } } } }
  reviews(last: 20) { nodes { author { login } submittedAt } }
  comments(last: 20) { nodes { author { login } createdAt } }
}
"""


class GitHubError(Exception):
    pass


class AuthError(GitHubError):
    pass


def search_queries(now):
    since = (now - timedelta(days=7)).date().isoformat()
    return {
        "review": "is:open is:pr review-requested:@me archived:false",
        "mine": "is:open is:pr author:@me archived:false",
        "merged": f"is:pr author:@me is:merged merged:>={since}",
    }


def fetch(token, now):
    body = json.dumps({"query": QUERY, "variables": search_queries(now)}).encode()
    request = urllib.request.Request(API_URL, data=body, headers={
        "Authorization": f"bearer {token}", "Content-Type": "application/json", "User-Agent": "bip",
    })
    # certifi: o Python empacotado nem sempre acha os certificados do sistema.
    context = ssl.create_default_context(cafile=certifi.where())
    try:
        with urllib.request.urlopen(request, timeout=30, context=context) as response:
            data = json.load(response)
    except urllib.error.HTTPError as e:
        if e.code == 401:
            raise AuthError("401") from e
        raise GitHubError(f"HTTP {e.code}") from e
    except urllib.error.URLError as e:
        raise GitHubError(str(e.reason)) from e
    if data.get("errors"):
        raise GitHubError(data["errors"][0].get("message", "GraphQL error"))
    return data["data"]


def _latest_feedback(node, viewer):
    times = [
        item.get("submittedAt") or item.get("createdAt")
        for conn in (node["reviews"], node["comments"])
        for item in conn["nodes"]
        if item.get("author") and item["author"]["login"] != viewer
    ]
    return max((t for t in times if t), default=None)


def _pr(node, viewer):
    commits = node["commits"]["nodes"]
    rollup = commits[0]["commit"]["statusCheckRollup"] if commits else None
    return PR(
        id=node["id"],
        repo=node["repository"]["nameWithOwner"],
        number=node["number"],
        title=node["title"],
        url=node["url"],
        author=(node.get("author") or {}).get("login", "ghost"),
        draft=node["isDraft"],
        created_at=node["createdAt"],
        merged_at=node.get("mergedAt"),
        head=node["headRefName"],
        base=node["baseRefName"],
        head_oid=node["headRefOid"],
        ci=rollup["state"] if rollup else None,
        review_decision=node.get("reviewDecision"),
        conflicting=node.get("mergeable") == "CONFLICTING",
        last_feedback_at=_latest_feedback(node, viewer),
    )


def parse(data):
    viewer = data["viewer"]["login"]

    def prs(key):
        return tuple(_pr(n, viewer) for n in data[key]["nodes"] if n and n.get("id"))

    return Snapshot(viewer=viewer, review=prs("review"), mine=prs("mine"), merged=prs("merged"))


def gh_cli_token():
    """Token do GitHub CLI (`gh auth token`), se instalado e logado."""
    exe = shutil.which("gh") or next((p for p in GH_PATHS if Path(p).exists()), None)
    if not exe:
        return None
    flags = {"creationflags": subprocess.CREATE_NO_WINDOW} if sys.platform == "win32" else {}
    try:
        result = subprocess.run([exe, "auth", "token"], capture_output=True, text=True, timeout=10, **flags)
    except (OSError, subprocess.TimeoutExpired):
        return None
    token = result.stdout.strip()
    return token if result.returncode == 0 and token else None
```

`stacks.py`:
```python
"""Pilhas de stacked PRs: quem é filho de quem e quem precisa de rebase."""

from collections import defaultdict


def analyze(mine, merged, memory):
    """Retorna (parent, rebase, nova_memoria).

    parent: id do PR -> id do pai aberto.
    rebase: id do PR -> número do pai mergeado, ou None quando o motivo é conflito.
    memória: id do filho -> {"parent_id", "parent_number", "oid", "parent_merged"?}. Guarda quem era o pai
    porque o GitHub costuma trocar a base do filho para main quando o pai é mergeado.
    """
    open_by_head = {(p.repo, p.head): p for p in mine}
    merged_by_head = {(p.repo, p.head): p for p in merged}
    merged_ids = {p.id for p in merged}
    parent, rebase, new_memory = {}, {}, {}

    for pr in mine:
        entry = memory.get(pr.id)
        open_parent = open_by_head.get((pr.repo, pr.base))
        if open_parent and open_parent.id != pr.id:
            parent[pr.id] = open_parent.id
            new_memory[pr.id] = {"parent_id": open_parent.id, "parent_number": open_parent.number, "oid": pr.head_oid}
        elif entry and pr.head_oid == entry["oid"]:
            # Pai não está mais aberto e o filho não recebeu push desde então.
            entry = {**entry, "parent_merged": entry.get("parent_merged") or entry["parent_id"] in merged_ids}
            new_memory[pr.id] = entry
            if entry["parent_merged"]:
                rebase[pr.id] = entry["parent_number"]

        merged_parent = merged_by_head.get((pr.repo, pr.base))
        if merged_parent:  # a base ainda aponta para a branch do pai já mergeado
            rebase[pr.id] = merged_parent.number
        if pr.id not in rebase and pr.conflicting:
            rebase[pr.id] = None

    return parent, rebase, new_memory


def ordered(mine, parent, key):
    """Lista (pr, profundidade) com cada filho logo abaixo do pai; raízes e irmãos ordenados por `key`."""
    ids = {p.id for p in mine}
    children, roots = defaultdict(list), []
    for pr in mine:
        p = parent.get(pr.id)
        (children[p] if p in ids else roots).append(pr)

    rows, seen = [], set()

    def walk(pr, depth):
        if pr.id in seen:
            return
        seen.add(pr.id)
        rows.append((pr, depth))
        for child in sorted(children[pr.id], key=key):
            walk(child, depth + 1)

    for root in sorted(roots, key=key):
        walk(root, 0)
    for pr in sorted(mine, key=key):  # PRs presos em ciclo (sem raiz) não podem sumir
        walk(pr, 0)
    return rows
```

```python
"""Compara dois estados e decide o que notificar."""

from dataclasses import dataclass

from model import FAILING, PR


@dataclass(frozen=True)
class Event:
    kind: str  # review_requested, ci_failed, approved, changes_requested, commented, parent_merged
    pr: PR
    parent: int | None = None  # número do pai mergeado (parent_merged)


def diff(prev, cur):
    if prev is None:
        return []  # primeira busca: só registra o estado atual
    events = []

    seen = {p.id for p in prev.snapshot.review}
    events += [Event("review_requested", p) for p in cur.snapshot.review if p.id not in seen]

    old_mine = {p.id: p for p in prev.snapshot.mine}
    for pr in cur.snapshot.mine:
        old = old_mine.get(pr.id)
        if pr.ci in FAILING and (old is None or old.ci not in FAILING):
            events.append(Event("ci_failed", pr))
        if old:
            if pr.review_decision != old.review_decision and pr.review_decision == "APPROVED":
                events.append(Event("approved", pr))
            elif pr.review_decision != old.review_decision and pr.review_decision == "CHANGES_REQUESTED":
                events.append(Event("changes_requested", pr))
            elif pr.last_feedback_at and pr.last_feedback_at != old.last_feedback_at:
                events.append(Event("commented", pr))
        parent = cur.rebase.get(pr.id)
        if parent is not None and prev.rebase.get(pr.id) != parent:
            events.append(Event("parent_merged", pr, parent))
    return events
```

```python
"""Monta o conteúdo do menu (texto puro; o tray.py transforma em itens do sistema)."""

from dataclasses import dataclass
from datetime import datetime

from i18n import ago, short_ref, t
from model import ORDER, status
from stacks import ordered

EMOJI = {"rebase": "🔁", "failing": "❌", "changes": "💬", "running": "🟡", "draft": "📝",
         "approved": "✅", "waiting": "⏳"}
MAX_PER_SECTION = 15
MAX_TITLE = 55


@dataclass(frozen=True)
class Entry:
    text: str
    url: str | None = None
    header: bool = False  # texto informativo, não clicável


def _title(text):
    return text if len(text) <= MAX_TITLE else text[:MAX_TITLE] + "…"


def _status_text(pr, state, lang):
    s = status(pr, state)
    if s == "rebase":
        parent = state.rebase[pr.id]
        return t("status_conflict", lang) if parent is None else t("status_rebase", lang, parent=parent)
    return t(f"status_{s}", lang)


def _limit(rows, lang):
    if len(rows) <= MAX_PER_SECTION:
        return rows
    return rows[:MAX_PER_SECTION] + [Entry(t("more", lang, n=len(rows) - MAX_PER_SECTION), header=True)]


def entries(state, now, lang):
    snap = state.snapshot
    out = [Entry(t("review_header", lang, n=len(snap.review)), header=True)]
    reviews = []
    for pr in sorted(snap.review, key=lambda p: p.created_at):  # o que espera há mais tempo primeiro
        age = ago((now - datetime.fromisoformat(pr.created_at)).total_seconds(), lang)
        reviews.append(Entry(f"👀 {short_ref(pr)}  {_title(pr.title)} — {pr.author} · {age}", url=pr.url))
    out += _limit(reviews, lang) or [Entry(t("no_reviews", lang), header=True)]

    out.append(Entry(t("mine_header", lang, n=len(snap.mine)), header=True))
    key = lambda pr: (ORDER.index(status(pr, state)), -pr.number)  # noqa: E731
    mine = []
    for pr, depth in ordered(snap.mine, state.parent, key):
        indent = "    " * depth + ("↳ " if depth else "")
        emoji = EMOJI[status(pr, state)]
        text = f"{indent}{emoji} {short_ref(pr)}  {_title(pr.title)} — {_status_text(pr, state, lang)}"
        mine.append(Entry(text, url=pr.url))
    out += _limit(mine, lang) or [Entry(t("no_prs", lang), header=True)]
    return out
```

```python
import shutil
import subprocess
import sys
from pathlib import Path

from PIL import Image, ImageDraw, ImageFilter

HERE = Path(__file__).resolve().parent
S = 4
SIZE = 1024 * S


def draw_icon():
    img = Image.new("RGBA", (SIZE, SIZE), (0, 0, 0, 0))
    margin, radius = 100 * S, 185 * S
    box = (margin, margin, SIZE - margin, SIZE - margin)

    shadow = Image.new("RGBA", (SIZE, SIZE), (0, 0, 0, 0))
    ImageDraw.Draw(shadow).rounded_rectangle((box[0], box[1] + 14 * S, box[2], box[3] + 14 * S), radius,
                                             fill=(0, 0, 0, 90))
    img.alpha_composite(shadow.filter(ImageFilter.GaussianBlur(22 * S)))

    top, bottom = (255, 150, 40), (240, 70, 110)
    gradient = Image.new("RGBA", (SIZE, SIZE))
    gd = ImageDraw.Draw(gradient)
    for y in range(box[1], box[3]):
        k = (y - box[1]) / (box[3] - box[1])
        gd.line((0, y, SIZE, y), fill=tuple(round(a + (b - a) * k) for a, b in zip(top, bottom)))
    mask = Image.new("L", (SIZE, SIZE), 0)
    ImageDraw.Draw(mask).rounded_rectangle(box, radius, fill=255)
    img.paste(gradient, (0, 0), mask)

    d = ImageDraw.Draw(img)
    cx, cy = 360 * S, 512 * S
    d.ellipse((cx - 70 * S, cy - 70 * S, cx + 70 * S, cy + 70 * S), fill="white")
    for r in (190, 320):
        d.arc((cx - r * S, cy - r * S, cx + r * S, cy + r * S), start=-48, end=48, fill="white", width=58 * S)
    return img.resize((1024, 1024), Image.LANCZOS)


def main():
    icon = draw_icon()
    icon.save(HERE / "icon.png")
    icon.save(HERE / "Bip.ico", sizes=[(16, 16), (24, 24), (32, 32), (48, 48), (64, 64), (128, 128), (256, 256)])

            icon.resize((size * 2, size * 2), Image.LANCZOS).save(iconset / f"icon_{size}x{size}@2x.png")
        subprocess.run(["iconutil", "-c", "icns", str(iconset), "-o", str(HERE / "Bip.icns")], check=True)
```

```python
"""Iniciar com o sistema: LaunchAgent no macOS, chave Run do registro no Windows."""

import plistlib
import sys
from pathlib import Path

LABEL = "io.github.railima.bip"
RUN_KEY = r"Software\Microsoft\Windows\CurrentVersion\Run"


def _command():
    if getattr(sys, "frozen", False):
        if sys.platform == "darwin":
            app = Path(sys.executable).resolve().parents[2]  # .../Bip.app/Contents/MacOS/Bip
            return ["/usr/bin/open", "-a", str(app)]
        return [sys.executable]
    return [sys.executable, str(Path(__file__).resolve().parent / "app.py")]


def launch_agent_path():
    return Path.home() / "Library" / "LaunchAgents" / f"{LABEL}.plist"


def is_enabled():
    if sys.platform == "darwin":
        return launch_agent_path().exists()
    if sys.platform == "win32":
        import winreg

        try:
            with winreg.OpenKey(winreg.HKEY_CURRENT_USER, RUN_KEY) as key:
                winreg.QueryValueEx(key, "Bip")
            return True
        except FileNotFoundError:
            return False
    return False


def set_enabled(enabled):
    if sys.platform == "darwin":
        path = launch_agent_path()
        if enabled:
            path.parent.mkdir(parents=True, exist_ok=True)
            path.write_bytes(plistlib.dumps({"Label": LABEL, "ProgramArguments": _command(), "RunAtLoad": True}))
        else:
            path.unlink(missing_ok=True)
    elif sys.platform == "win32":
        import winreg

        with winreg.OpenKey(winreg.HKEY_CURRENT_USER, RUN_KEY, 0, winreg.KEY_SET_VALUE) as key:
            if enabled:
                winreg.SetValueEx(key, "Bip", 0, winreg.REG_SZ, " ".join(f'"{p}"' for p in _command()))
            else:
                try:
                    winreg.DeleteValue(key, "Bip")
                except FileNotFoundError:
                    pass
```

```python
"""Notificações nativas. Windows: winotify (clique abre o PR). macOS: UserNotifications quando empacotado
(clique abre o PR); fora do .app, osascript (sem clique)."""

import subprocess
import sys
import uuid
import webbrowser
from pathlib import Path

ICON = Path(getattr(sys, "_MEIPASS", Path(__file__).resolve().parent)) / "icon" / "icon.png"
_center = None  # UNUserNotificationCenter, configurado em setup()


def setup():
    """Chamar uma vez, na thread principal, com o app já rodando."""
    global _center
    if sys.platform != "darwin" or not getattr(sys, "frozen", False):
        return
    try:
        import objc
        from Foundation import NSObject
        from UserNotifications import (
            UNAuthorizationOptionAlert,
            UNAuthorizationOptionSound,
            UNNotificationPresentationOptionBanner,
            UNNotificationPresentationOptionList,
            UNUserNotificationCenter,
        )

        class Delegate(NSObject):
            def userNotificationCenter_didReceiveNotificationResponse_withCompletionHandler_(self, c, response, done):
                url = response.notification().request().content().userInfo().get("url")
                if url:
                    webbrowser.open(url)
                done()

            def userNotificationCenter_willPresentNotification_withCompletionHandler_(self, c, n, done):
                done(UNNotificationPresentationOptionBanner | UNNotificationPresentationOptionList)

        center = UNUserNotificationCenter.currentNotificationCenter()
        setup.delegate = Delegate.alloc().init()  # referência forte: o centro guarda o delegate como weak
        center.setDelegate_(setup.delegate)
        center.requestAuthorizationWithOptions_completionHandler_(
            UNAuthorizationOptionAlert | UNAuthorizationOptionSound, lambda granted, error: None)
        _center = center
    except Exception:  # sem bundle/permissão: cai no osascript
        _center = None


def notify(title, body, url=None):
    if sys.platform == "win32":
        from winotify import Notification

        toast = Notification(app_id="Bip", title=title, msg=body, launch=url or "",
                             icon=str(ICON) if ICON.exists() else "")
        toast.show()
    elif sys.platform == "darwin":
        if _center is not None:
            from UserNotifications import UNMutableNotificationContent, UNNotificationRequest, UNNotificationSound

            content = UNMutableNotificationContent.alloc().init()
            content.setTitle_(title)
            content.setBody_(body)
            content.setSound_(UNNotificationSound.defaultSound())
            content.setUserInfo_({"url": url or ""})
            request = UNNotificationRequest.requestWithIdentifier_content_trigger_(str(uuid.uuid4()), content, None)
            _center.addNotificationRequest_withCompletionHandler_(request, None)
        else:
            script = "display notification item 2 of argv with title item 1 of argv"
            subprocess.run(["osascript", "-e", f"on run argv\n{script}\nend run", title, body], check=False)
```

```python
"""Ícone na barra de menu/bandeja e o menu (pystray)."""

import sys
import time
import webbrowser
from datetime import datetime, timezone

import pystray
from pystray import Menu, MenuItem

import autostart
import config
from github_api import AuthError
from i18n import LANGUAGES, ago, t
from icons import tray_icon
from menu import entries
from model import attention_count

INTERVALS = (1, 2, 5, 10)


def _on_main_thread(fn):
    """No macOS, mexer no menu/ícone fora da thread principal pode travar o app."""
    if sys.platform == "darwin":
        from PyObjCTools import AppHelper

        AppHelper.callAfter(fn)
    else:
        fn()


class Tray:
    def __init__(self, app):
        self.app = app
        self.icon = pystray.Icon("bip", tray_icon(0, False), "Bip", menu=Menu(self._items))

    # ---------- Menu ----------

    def _lang(self):
        return config.setting("ui_language")

    def _items(self):
        lang, app = self._lang(), self.app
        if app.error:
            kind, detail = app.error
            text = t("error_auth", lang) if kind == "auth" else t("error_network", lang, detail=detail)
            yield MenuItem(text, None, enabled=False)
            yield Menu.SEPARATOR
        if app.state:
            for entry in entries(app.state, datetime.now(timezone.utc), lang):
                if entry.header:
                    yield MenuItem(entry.text, None, enabled=False)
                else:
                    yield MenuItem(entry.text, lambda _i, _m, url=entry.url: webbrowser.open(url))
        else:
            yield MenuItem(t("loading", lang), None, enabled=False)
        yield Menu.SEPARATOR

        if app.updated_at:
            age = time.time() - app.updated_at
            label = t("updated_now", lang) if age < 60 else t("updated", lang, ago=ago(age, lang))
            yield MenuItem(label, None, enabled=False)
        yield MenuItem(t("refresh", lang), lambda: app.refresh_now())
        if app.paused():
            yield MenuItem(t("resume", lang), lambda: self._pause(False))
        else:
            yield MenuItem(t("pause", lang), lambda: self._pause(True))
        yield MenuItem(t("settings", lang), Menu(*self._settings(lang)))
        yield MenuItem(t("quit", lang), lambda: self.app.quit())

    def _settings(self, lang):
        interval = Menu(*(
            MenuItem(t("minutes", lang, n=n), lambda _i, _m, n=n: self._set(interval=n),
                     checked=lambda _m, n=n: config.setting("interval") == n, radio=True)
            for n in INTERVALS
        ))
        language = Menu(*(
            MenuItem(name, lambda _i, _m, code=code: self._set(ui_language=code),
                     checked=lambda _m, code=code: config.setting("ui_language") == code, radio=True)
            for code, name in LANGUAGES.items()
        ))
        return (
            MenuItem(t("interval", lang), interval),
            MenuItem(t("language", lang), language),
            MenuItem(t("token", lang), lambda: self.app.ask_token()),
            MenuItem(t("start_at_login", lang), lambda: self._toggle_autostart(),
                     checked=lambda _m: autostart.is_enabled()),
        )

    def _set(self, **values):
        config.save_config(**values)
        self.app.refresh_now()

    def _pause(self, paused):
        config.save_config(paused_until=time.time() + 3600 if paused else 0)
        self.update()

    def _toggle_autostart(self):
        autostart.set_enabled(not autostart.is_enabled())
        self.update()

    # ---------- Ícone ----------

    def update(self):
        app = self.app
        count = attention_count(app.state) if app.state else 0
        error = app.error is not None

        def apply():
            # macOS: número em texto ao lado do ícone; Windows: desenhado no ícone.
            self.icon.icon = tray_icon(0 if sys.platform == "darwin" else count, error)
            self.icon.title = f"Bip — {count}" if count else "Bip"
            self.icon.update_menu()
            if sys.platform == "darwin":
                item = getattr(self.icon, "_status_item", None)
                if item is not None:
                    item.button().setTitle_(f" {count}" if count else "")

        _on_main_thread(apply)

    def run(self, on_ready):
        def setup(icon):
            icon.visible = True
            on_ready()

        self.icon.run(setup=setup)

    def stop(self):
        self.icon.stop()
```

```python
"""Bip: painel de PRs do GitHub na barra de menu/bandeja."""

import subprocess
import sys
import threading
import time
from datetime import datetime, timezone

import config
import notifier
from events import diff
from github_api import AuthError, GitHubError, fetch, gh_cli_token, parse
from i18n import notification, t
from model import State
from stacks import analyze
from tray import Tray, _on_main_thread


class Bip:
    def __init__(self):
        self.state = None
        self.error = None  # ("auth", None) ou ("network", detalhe)
        self.updated_at = None
        self.memory = config.load_memory()
        self.wake = threading.Event()
        self.running = True
        self.tray = Tray(self)

    def paused(self):
        return config.setting("paused_until") > time.time()

    def refresh_now(self):
        self.wake.set()

    def refresh_once(self):
        token = config.load_token() or gh_cli_token()
        if not token:
            raise AuthError("no token")
        snapshot = parse(fetch(token, datetime.now(timezone.utc)))
        parent, rebase, self.memory = analyze(snapshot.mine, snapshot.merged, self.memory)
        config.save_memory(self.memory)
        state = State(snapshot, parent, rebase)
        events = diff(self.state, state)
        self.state, self.error, self.updated_at = state, None, time.time()
        if not self.paused():
            lang = config.setting("ui_language")
            for event in events:
                title, body = notification(event, lang)
                notifier.notify(title, body, event.pr.url)

    def loop(self):
        failures = 0
        while self.running:
            try:
                self.refresh_once()
                failures = 0
            except AuthError:
                self.error, failures = ("auth", None), failures + 1
            except GitHubError as e:
                self.error, failures = ("network", str(e)), failures + 1
            self.tray.update()
            delay = config.setting("interval") * 60 if not failures else min(600, 30 * 2 ** (failures - 1))
            deadline = time.time() + delay
            # Acorda a cada minuto para atualizar "Atualizado há X" no menu.
            while self.running and time.time() < deadline:
                if self.wake.wait(min(60, deadline - time.time())):
                    break
                self.tray.update()
            self.wake.clear()

    def ask_token(self):
        """Caixa de diálogo nativa para colar o token; vazio volta a usar o login do gh."""
        prompt = t("token_prompt", config.setting("ui_language"))
        if sys.platform == "darwin":
            script = ('on run argv\ndisplay dialog (item 1 of argv) default answer "" with hidden answer '
                      'with title "Bip"\nreturn text returned of result\nend run')
            result = subprocess.run(["osascript", "-e", script, prompt], capture_output=True, text=True)
        else:
            script = ("Add-Type -AssemblyName Microsoft.VisualBasic; "
                      "[Microsoft.VisualBasic.Interaction]::InputBox($args[0], 'Bip')")
            result = subprocess.run(["powershell", "-NoProfile", "-Command", script, prompt],
                                    capture_output=True, text=True, creationflags=subprocess.CREATE_NO_WINDOW)
        if result.returncode == 0:  # cancelar não altera nada
            config.save_token(result.stdout.strip())
            self.refresh_now()

    def quit(self):
        self.running = False
        self.wake.set()
        self.tray.stop()

    def run(self):
        def ready():
            _on_main_thread(notifier.setup)
            threading.Thread(target=self.loop, daemon=True).start()

        self.tray.run(ready)


if __name__ == "__main__":
    Bip().run()
```
